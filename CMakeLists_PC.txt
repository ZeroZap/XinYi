# XinYi Framework - PC Platform Build
cmake_minimum_required(VERSION 3.12)
project(XY_Framework_PC VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)

# PC Platform
add_definitions(-DHAL_PLATFORM_PC)
add_definitions(-DCONFIG_PLATFORM_PC)

# Include directories
include_directories(
    components/hal/inc
    components/hal/PC
    components/device/inc
    components/trace/xy_log/inc
    components/clib/xy_clib/inc
    components/kernel/osal/inc
    components/kernel/osal/backend/baremetal
    components/sensor/inc
    components/crypto/inc
    components/dm/inc
)

# Sources
set(SOURCES
    components/hal/PC/xy_hal_pc.c
    components/hal/PC/xy_hal_pin_pc.c
    components/hal/PC/xy_hal_i2c_pc.c
    components/hal/PC/xy_hal_spi_pc.c
    components/hal/PC/xy_hal_uart_pc.c
    components/device/src/xy_device.c
    components/trace/xy_log/src/xy_log.c
    components/kernel/osal/backend/baremetal/xy_os_baremetal.c
    components/sensor/src/xy_dht11.c
    components/crypto/src/xy_crc.c
)

# Create library
add_library(xinyi STATIC ${SOURCES})

# Example
add_executable(test_pc examples/pc_all_components/src/main.c)
target_link_libraries(test_pc xinyi)

message(STATUS "✅ PC Platform Build Ready")
